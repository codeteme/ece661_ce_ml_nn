import matplotlib.pyplot as plt

W0 = np.array([-0.2, 0.0, 0.45]).reshape(-1, 1)
current_W = W0.copy()

r = 0.005
K = my_X.shape[0]
J_history = []

for epoch in range(20):
    for k in range(K):
        x_k = my_X[k:k+1, :].T
        d_k = my_D[k, 0]      
        e_k = d_k - (x_k.T @ current_W).item()
        current_W = current_W + r * e_k * x_k

    y = my_X @ current_W
    e = my_D - y
    J = (1/(2*K)) * (e.T @ e)
    J_history.append(J.item())

print("Final W after 20 epochs:", current_W)

plt.figure(figsize=(10, 5))
plt.plot(range(1, 21), J_history, marker='o')
plt.yscale('log')
plt.xlabel('Epoch')
plt.ylabel('MSE Loss')
plt.title('LMS Convergence (r = 0.005)')
plt.grid(True, which='both', alpha=0.3)
plt.show()

def learning_rate_weight_update(W_init, r):
    current_W = W_init.copy()
    J_history = []
    K = my_X.shape[0]

    for epoch in range(20):
        for k in range(K):
            x_k = my_X[k:k+1, :].T
            d_k = my_D[k, 0]      
            e_k = d_k - (x_k.T @ current_W).item()
            current_W = current_W + r * e_k * x_k

        y = my_X @ current_W
        e = my_D - y
        J = (1/(2*K)) * (e.T @ e)
        J_history.append(J.item())

    return J_history, current_W
